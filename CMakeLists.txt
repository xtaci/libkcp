cmake_minimum_required(VERSION 3.10)
project(kcp LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(KCP_ENABLE_CRYPTO "Enable kcp-go compatible encryption using LibTomCrypt" OFF)
add_library(kcp STATIC src/ikcp.c src/sess.c src/fec.c src/reedsolomon.c src/autotune.c
                       src/galois.c src/galois_table.c)
# The upstream byte-wise branch emits little-endian wire values on any host and
# avoids unaligned integer accesses without modifying the upstream KCP files.
set_source_files_properties(src/ikcp.c PROPERTIES COMPILE_DEFINITIONS IWORDS_BIG_ENDIAN=1)
target_include_directories(kcp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(WIN32)
    target_compile_definitions(kcp PRIVATE _WIN32_WINNT=0x0600)
    target_link_libraries(kcp PUBLIC ws2_32 bcrypt)
endif()
if(KCP_ENABLE_CRYPTO)
    find_path(TOMCRYPT_INCLUDE_DIR tomcrypt.h)
    find_library(TOMCRYPT_LIBRARY NAMES tomcrypt)
    if(NOT TOMCRYPT_INCLUDE_DIR OR NOT TOMCRYPT_LIBRARY)
        message(FATAL_ERROR "KCP_ENABLE_CRYPTO requires LibTomCrypt headers and library")
    endif()
    target_sources(kcp PRIVATE src/crypt.c src/crypt_backend.c)
    target_include_directories(kcp SYSTEM PRIVATE ${TOMCRYPT_INCLUDE_DIR})
    target_compile_definitions(kcp PUBLIC KCP_ENABLE_CRYPTO)
    target_link_libraries(kcp PUBLIC ${TOMCRYPT_LIBRARY})
    if(NOT WIN32)
        find_package(Threads REQUIRED)
        target_link_libraries(kcp PUBLIC Threads::Threads)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
